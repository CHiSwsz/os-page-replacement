cmake_minimum_required(VERSION 3.16)
project(page_replacement LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(paging src/paging.cpp src/adaptive.cpp)
target_include_directories(paging PUBLIC include)
target_compile_options(paging PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(pager src/main.cpp)
target_link_libraries(pager PRIVATE paging)
target_compile_options(pager PRIVATE -Wall -Wextra -Wpedantic -Wconversion)

include(CTest)
if(BUILD_TESTING)
  add_executable(paging_tests tests/test_paging.cpp)
  target_link_libraries(paging_tests PRIVATE paging)
  target_compile_options(paging_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  add_test(NAME algorithms COMMAND paging_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:pager>)
endif()

if(ENABLE_SANITIZERS)
  foreach(target paging pager)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
  if(BUILD_TESTING)
    target_compile_options(paging_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
